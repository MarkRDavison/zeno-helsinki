CMAKE_MINIMUM_REQUIRED(VERSION 3.15)
PROJECT(helsinki LANGUAGES C CXX)

if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
	set(HELSINKI_IS_TOP_LEVEL ON)
else()
	set(HELSINKI_IS_TOP_LEVEL OFF)
endif()

SET(HELSINKI_CXX_STANDARD 23)

SET(CMAKE_CXX_STANDARD ${HELSINKI_CXX_STANDARD})
SET(CMAKE_CXX_STANDARD_REQUIRED ON)

option(HELSINKI_BUILD_APPS "whether or not apps should be built" ${HELSINKI_IS_TOP_LEVEL})
option(HELSINKI_BUILD_TESTS "whether or not tests should be built" ${HELSINKI_IS_TOP_LEVEL})
OPTION(HELSINKI_STRICT "Enable strict compile options" ON)
option(HELSINKI_WARNING_AS_ERROR "Treat Helsinki compile warnings as errors" ON)
option(HELSINKI_ENABLE_ASAN "Enable AddressSanitizer on Helsinki targets" OFF)
option(HELSINKI_ENABLE_TRACY "Enable Tracy Profiler" OFF)

# Do not apply Werror to FetchContent deps. Helsinki targets use HELSINKI_WARNING_AS_ERROR.
set(CMAKE_COMPILE_WARNING_AS_ERROR OFF)

if(CMAKE_BUILD_TYPE STREQUAL "Debug")
	set(HELSINKI_ENABLE_TRACY ON)
endif()

if(HELSINKI_IS_TOP_LEVEL AND BUILD_SHARED_LIBS)
	set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
	set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
	set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
endif()

LIST(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

if (CMAKE_VERSION VERSION_GREATER_EQUAL 3.19)
  CMAKE_POLICY(SET CMP0110 NEW)
endif()

INCLUDE("${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/CMakeLists.txt")
ADD_SUBDIRECTORY(lib)

if(HELSINKI_BUILD_APPS)
	ADD_SUBDIRECTORY(app)
endif()
if(HELSINKI_BUILD_TESTS)
	ENABLE_TESTING()
	ADD_SUBDIRECTORY(test)
endif()
